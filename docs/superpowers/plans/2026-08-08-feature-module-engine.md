# Feature-Module Engine Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the monolithic string-builder scaffolder with a config-driven feature-module engine that copies only enabled feature folders, regenerates shared wiring via tool-owned aggregators, and syncs an existing project in place (add + remove) from `forge.config.yaml`.

**Architecture:** Feature = a folder under `templates/features/<id>/` with a `feature.yaml` manifest + a `files/` tree it owns exclusively. A sync engine (`tools/scaffold/*.mjs`) reads `forge.config.yaml`, copies/deletes owned folders, regenerates `*.generated.*` aggregator files from each enabled feature's `contributes:` block, reconciles `package.json` deps by ownership, and records applied state in `.forge/state.json`. Pure functions (render, aggregate, deps) are unit-tested; the IO orchestrator (sync) is covered by temp-dir end-to-end tests.

**Tech Stack:** Node ≥20 ESM (`.mjs`), zero runtime deps (regex token substitution + the repo's existing minimal YAML parser + Node `crypto`/`fs`), Vitest for tests.

## Global Constraints

- Zero runtime dependencies in `tools/scaffold/**` — no npm template engine or YAML package. Only Node built-ins + the in-repo YAML parser.
- Node ESM `.mjs` modules only; named exports.
- Tokens are an explicit closed set: `{{projectName}}`, `{{apiBaseUrl}}`, `{{timeout}}`, plus per-feature option tokens declared in that feature's `feature.yaml`.
- All emitted TypeScript targets the repo stack: React 19, TS 7 strict, Vite 8, React Router v7, React Query v5, SCSS modules.
- Every generated aggregator file starts with the banner line: `// AUTO-GENERATED by scaffold.mjs — do not edit. Managed by forge.config.yaml.`
- English only in all code, comments, docs (repo Language Rule).
- Coverage floor 85% (lines/functions/branches/statements) for `tools/scaffold/**`, enforced by the Vitest config already in the repo.
- Conventional Commits; end commit messages with `Co-Authored-By: Claude Opus 4.8 <noreply@anthropic.com>`.
- Sync NEVER runs a package-manager install — it only edits files and prints a reminder.

```js
import { existsSync, readdirSync, statSync } from "node:fs";
import { join, relative, sep } from "node:path";

export function renderTokens(content, tokens) {
  const out = content.replace(/\{\{(\w+)\}\}/g, (match, key) => {
    if (!(key in tokens)) throw new Error(`Unknown token: {{${key}}}`);
    return tokens[key];
  });
  return out;
}

export function listFiles(dir) {
  if (!existsSync(dir)) return [];
  const walk = (d) =>
    readdirSync(d).flatMap((name) => {
      const full = join(d, name);
      return statSync(full).isDirectory() ? walk(full) : [full];
    });
  return walk(dir)
    .map((f) => relative(dir, f).split(sep).join("/"))
    .sort();
}
```

```js
import { existsSync, readdirSync, readFileSync, statSync } from "node:fs";
import { join } from "node:path";
import { parseYaml } from "./yaml.mjs";

const EMPTY_CONTRIB = { routes: [], nav: [], providers: [], http: [] };

export function loadCatalog(featuresDir) {
  if (!existsSync(featuresDir)) return [];
  return readdirSync(featuresDir)
    .filter((name) => statSync(join(featuresDir, name)).isDirectory())
    .map((name) => {
      const manifestPath = join(featuresDir, name, "feature.yaml");
      const m = existsSync(manifestPath)
        ? parseYaml(readFileSync(manifestPath, "utf8"))
        : {};
      return {
        id: m.id ?? name,
        name: m.name ?? name,
        description: m.description ?? "",
        default: m.default ?? false,
        deps: m.deps ?? [],
        devDeps: m.devDeps ?? [],
        options: m.options ?? {},
        contributes: { ...EMPTY_CONTRIB, ...(m.contributes ?? {}) },
      };
    })
    .sort((a, b) => a.id.localeCompare(b.id));
}
```

```js
import { existsSync, readFileSync, writeFileSync } from "node:fs";
import { parseYaml } from "./yaml.mjs";

export function readConfig(path) {
  if (!existsSync(path)) throw new Error(`forge.config.yaml not found at ${path}`);
  const c = parseYaml(readFileSync(path, "utf8"));
  return { project: c.project, packageManager: c.packageManager, features: c.features ?? {} };
}

export function enabledFeatureIds(config) {
  return Object.entries(config.features ?? {})
    .filter(([, v]) => v === true || (v && typeof v === "object"))
    .map(([k]) => k)
    .sort();
}

export function validateConfig(config, availableIds) {
  const unknown = enabledFeatureIds(config).filter((id) => !availableIds.includes(id));
  if (unknown.length) {
    throw new Error(
      `Unknown feature(s) in config: ${unknown.join(", ")}. Valid: ${availableIds.join(", ")}`,
    );
  }
}

export function writeConfig(path, config) {
  const lines = [`project: ${config.project}`, `packageManager: ${config.packageManager}`, "features:"];
  for (const id of Object.keys(config.features).sort()) {
    const v = config.features[id];
    if (v && typeof v === "object") {
      lines.push(`  ${id}:`);
      for (const [ok, ov] of Object.entries(v)) {
        lines.push(`    ${ok}: ${Array.isArray(ov) ? `[${ov.join(", ")}]` : ov}`);
      }
    } else {
      lines.push(`  ${id}: ${v ? "true" : "false"}`);
    }
  }
  writeFileSync(path, lines.join("\n") + "\n");
}
```

```js
const BANNER =
  "// AUTO-GENERATED by scaffold.mjs — do not edit. Managed by forge.config.yaml.\n";

function importName(component, id, kind) {
  return `${component}_${id}_${kind}`;
}

function buildRoutes(features) {
  const items = features.flatMap((f) =>
    f.contributes.routes.map((r) => ({ ...r, alias: importName(r.component, f.id, "route") })),
  );
  const imports = items
    .map((r) => `import { ${r.component} as ${r.alias} } from "${r.import}";`)
    .join("\n");
  const entries = items
    .map((r) => `  { path: "${r.path}", element: <${r.alias} /> },`)
    .join("\n");
  return `${BANNER}import type { RouteObject } from "react-router-dom";
${imports}

export const featureRoutes: RouteObject[] = [
${entries}
];
`;
}

function buildNav(features) {
  const entries = features
    .flatMap((f) => f.contributes.nav)
    .map((n) => `  { label: "${n.label}", to: "${n.to}" },`)
    .join("\n");
  return `${BANNER}export type NavItem = { label: string; to: string };

export const featureNav: NavItem[] = [
${entries}
];
`;
}

function buildProviders(features) {
  const items = features.flatMap((f) =>
    f.contributes.providers.map((p) => ({ ...p, alias: importName(p.component, f.id, "provider") })),
  );
  const imports = items
    .map((p) => `import { ${p.component} as ${p.alias} } from "${p.import}";`)
    .join("\n");
  const open = items.map((p) => `<${p.alias}>`).join("");
  const close = items.map((p) => `</${p.alias}>`).reverse().join("");
  return `${BANNER}import type { ReactNode } from "react";
${imports}

export function FeatureProviders({ children }: { children: ReactNode }) {
  return ${open || "<>"}{children}${close || "</>"};
}
`;
}

function buildHttpHooks(features) {
  const handlers = features.flatMap((f) => f.contributes.http);
  const imports = handlers
    .map((h, i) => `import { ${Object.values(h)[0].split("/").pop()} as h${i} } from "${Object.values(h)[0]}";`)
    .join("\n");
  const list = handlers.map((_, i) => `h${i}`).join(", ");
  return `${BANNER}${imports}

export const responseErrorHandlers = [${list}];
`;
}

export function buildAggregators(enabledFeatures) {
  const f = [...enabledFeatures].sort((a, b) => a.id.localeCompare(b.id));
  return {
    "src/app/routes.generated.tsx": buildRoutes(f),
    "src/app/nav.generated.ts": buildNav(f),
    "src/app/providers.generated.tsx": buildProviders(f),
    "src/api/http/hooks.generated.ts": buildHttpHooks(f),
  };
}
```

```js
import { createHash } from "node:crypto";
import { existsSync, mkdirSync, readFileSync, writeFileSync } from "node:fs";
import { dirname, join } from "node:path";

const statePath = (projectDir) => join(projectDir, ".forge", "state.json");

export function hashContent(text) {
  return createHash("sha256").update(text).digest("hex");
}

export function readState(projectDir) {
  const p = statePath(projectDir);
  if (!existsSync(p)) return { features: {} };
  return JSON.parse(readFileSync(p, "utf8"));
}

export function writeState(projectDir, state) {
  const p = statePath(projectDir);
  mkdirSync(dirname(p), { recursive: true });
  writeFileSync(p, JSON.stringify(state, Object.keys(state).sort(), 2) + "\n");
}
```

> Note: `JSON.stringify(state, replacerArray, 2)` with a key-array replacer only whitelists top-level keys. For deterministic deep output, instead sort keys when constructing `state` in `sync.mjs`; keep `writeState` as `JSON.stringify(state, null, 2)`. Use the `null` replacer form in the implementation.

```js
export function splitPackage(spec) {
  const at = spec.lastIndexOf("@");
  return at > 0 ? [spec.slice(0, at), spec.slice(at + 1)] : [spec, "latest"];
}

function sortObject(obj) {
  return Object.fromEntries(Object.entries(obj).sort(([a], [b]) => a.localeCompare(b)));
}

export function reconcileDeps(pkg, enabledFeatures, prevState) {
  const enabledIds = new Set(enabledFeatures.map((f) => f.id));
  const stillRequired = new Set();
  for (const f of enabledFeatures) {
    for (const spec of [...(f.deps ?? []), ...(f.devDeps ?? [])]) stillRequired.add(splitPackage(spec)[0]);
  }
  // add enabled
  for (const f of enabledFeatures) {
    for (const spec of f.deps ?? []) { const [n, r] = splitPackage(spec); pkg.dependencies[n] = r; }
    for (const spec of f.devDeps ?? []) { const [n, r] = splitPackage(spec); pkg.devDependencies[n] = r; }
  }
  // remove deps owned only by now-disabled features
  for (const [id, rec] of Object.entries(prevState.features ?? {})) {
    if (enabledIds.has(id)) continue;
    for (const spec of rec.deps ?? []) {
      const [n] = splitPackage(spec);
      if (!stillRequired.has(n)) delete pkg.dependencies[n];
    }
    for (const spec of rec.devDeps ?? []) {
      const [n] = splitPackage(spec);
      if (!stillRequired.has(n)) delete pkg.devDependencies[n];
    }
  }
  pkg.dependencies = sortObject(pkg.dependencies);
  pkg.devDependencies = sortObject(pkg.devDependencies);
  return pkg;
}
```

8. Rebuild `state.features` for the enabled set (files+hashes+deps) and `writeState`.
